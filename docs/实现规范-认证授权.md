# 38. Authentication / Authorization 完整实现

> 本章节提供双认证模型（Token + AppSecret 签名）的完整实现设计、中间件和校验流程。

## 38.1 认证中间件

```csharp
namespace CryptoPlatform.Api.Middleware;

public sealed class AuthenticationMiddleware
{
    private readonly RequestDelegate _next;
    private readonly ILogger<AuthenticationMiddleware> _logger;

    public AuthenticationMiddleware(RequestDelegate next, ILogger<AuthenticationMiddleware> logger)
    {
        _next = next;
        _logger = logger;
    }

    public async Task InvokeAsync(HttpContext context)
    {
        var path = context.Request.Path.Value ?? "";

        // 管理 API：/api/v1/admin/** → Token 认证
        if (path.StartsWithSegments("/api/v1/admin"))
        {
            await AuthenticateAdminAsync(context);
            return;
        }

        // 业务 API：/api/v1/crypto/** 和 /api/v1/keys/** → AppSecret 签名认证
        if (path.StartsWithSegments("/api/v1/crypto") || path.StartsWithSegments("/api/v1/keys"))
        {
            await AuthenticateAppAsync(context);
            return;
        }

        // 公开接口：健康检查等
        await _next(context);
    }

    private async Task AuthenticateAdminAsync(HttpContext context)
    {
        var authHeader = context.Request.Headers["Authorization"].FirstOrDefault();
        if (string.IsNullOrEmpty(authHeader) || !authHeader.StartsWith("Bearer "))
        {
            context.Response.StatusCode = 401;
            await context.Response.WriteAsJsonAsync(new { code = "AUTH_TOKEN_MISSING", message = "缺少认证令牌" });
            return;
        }

        var token = authHeader["Bearer ".Length..].Trim();
        var tokenService = context.RequestServices.GetRequiredService<IAdminTokenService>();
        var principal = await tokenService.ValidateAsync(token);

        if (principal is null)
        {
            context.Response.StatusCode = 401;
            await context.Response.WriteAsJsonAsync(new { code = "AUTH_TOKEN_INVALID", message = "令牌无效或已过期" });
            return;
        }

        context.Items["OperatorType"] = "ADMIN";
        context.Items["OperatorId"] = principal.OperatorId;
        context.Items["Role"] = principal.Role;
        context.Items["Scopes"] = principal.Scopes;

        await _next(context);
    }

    private async Task AuthenticateAppAsync(HttpContext context)
    {
        var appId = context.Request.Headers["X-App-Id"].FirstOrDefault();
        var timestamp = context.Request.Headers["X-Timestamp"].FirstOrDefault();
        var nonce = context.Request.Headers["X-Nonce"].FirstOrDefault();
        var signature = context.Request.Headers["X-Signature"].FirstOrDefault();

        if (string.IsNullOrEmpty(appId) || string.IsNullOrEmpty(signature))
        {
            context.Response.StatusCode = 401;
            await context.Response.WriteAsJsonAsync(new { code = "AUTH_CREDENTIALS_MISSING", message = "缺少认证凭据" });
            return;
        }

        var appAuthService = context.RequestServices.GetRequiredService<IAppAuthenticationService>();
        var result = await appAuthService.ValidateAsync(appId, timestamp, nonce, signature, context.Request);

        if (!result.Success)
        {
            context.Response.StatusCode = 401;
            await context.Response.WriteAsJsonAsync(new { code = result.ErrorCode, message = result.ErrorMessage });
            return;
        }

        context.Items["OperatorType"] = "APP";
        context.Items["AppId"] = appId;

        await _next(context);
    }
}
```

## 38.2 管理员 Token 服务

```csharp
namespace CryptoPlatform.Authentication;

public interface IAdminTokenService
{
    Task<AdminLoginResponse> LoginAsync(string username, string password, CancellationToken ct);
    Task<AdminPrincipal?> ValidateAsync(string token);
    Task RevokeAsync(string jti, CancellationToken ct);
}

public sealed record AdminLoginResponse(string AccessToken, string TokenType, int ExpiresIn, string OperatorId, string Role);

public sealed record AdminPrincipal(string OperatorId, string Username, string Role, IReadOnlyList<string> Scopes);

public sealed class AdminTokenService : IAdminTokenService
{
    private readonly CryptoPlatformDbContext _db;
    private readonly ITokenGenerator _tokenGen;
    private readonly IDistributedCache _redis;
    private readonly JwtSettings _jwtSettings;
    private readonly ILogger<AdminTokenService> _logger;

    public AdminTokenService(
        CryptoPlatformDbContext db, ITokenGenerator tokenGen,
        IDistributedCache redis, IOptions<JwtSettings> jwtSettings,
        ILogger<AdminTokenService> logger)
    {
        _db = db; _tokenGen = tokenGen; _redis = redis;
        _jwtSettings = jwtSettings.Value; _logger = logger;
    }

    public async Task<AdminLoginResponse> LoginAsync(string username, string password, CancellationToken ct)
    {
        var user = await _db.Users
            .Include(u => u.UserRoles).ThenInclude(ur => ur.RoleId)
            .FirstOrDefaultAsync(u => u.Username == username, ct)
            ?? throw new BusinessException("AUTH_LOGIN_FAILED", "用户名或密码错误");

        if (user.Status != 0)
            throw new BusinessException("AUTH_ACCOUNT_DISABLED", "账号已禁用");

        if (user.LockedUntil.HasValue && user.LockedUntil > DateTime.UtcNow)
            throw new BusinessException("AUTH_ACCOUNT_LOCKED", "账号已锁定");

        if (!VerifyPassword(password, user.PasswordHash))
        {
            user.LoginFailCount++;
            if (user.LoginFailCount >= 5)
                user.LockedUntil = DateTime.UtcNow.AddMinutes(30);
            await _db.SaveChangesAsync(ct);
            throw new BusinessException("AUTH_LOGIN_FAILED", "用户名或密码错误");
        }

        // 登录成功
        user.LoginFailCount = 0;
        user.LockedUntil = null;
        user.LastLoginAt = DateTime.UtcNow;
        await _db.SaveChangesAsync(ct);

        var role = "SYSTEM_ADMIN"; // TODO: 从 UserRoles 获取
        var scopes = GetScopesForRole(role);

        var token = _tokenGen.Generate(new TokenPayload(
            Issuer: _jwtSettings.Issuer,
            Subject: user.Id.ToString(),
            Audience: "management-api",
            OperatorId: user.Id.ToString(),
            Role: role,
            Scopes: scopes,
            ExpiresIn: _jwtSettings.ExpiresInSeconds));

        return new AdminLoginResponse(token, "Bearer", _jwtSettings.ExpiresInSeconds, user.Id.ToString(), role);
    }

    public async Task<AdminPrincipal?> ValidateAsync(string token)
    {
        var payload = _tokenGen.Validate(token);
        if (payload is null) return null;

        // 检查是否已撤销
        var revoked = await _redis.GetStringAsync($"token:revoked:{payload.Jti}");
        if (revoked is not null) return null;

        return new AdminPrincipal(payload.OperatorId, "", payload.Role, payload.Scopes);
    }

    public async Task RevokeAsync(string jti, CancellationToken ct)
    {
        await _redis.SetStringAsync($"token:revoked:{jti}", "1",
            new DistributedCacheEntryOptions { AbsoluteExpirationRelativeToNow = TimeSpan.FromHours(2) });
    }

    private static bool VerifyPassword(string password, string hash)
        => BCrypt.Net.BCrypt.Verify(password, hash); // 或自定义 SM3 哈希

    private static IReadOnlyList<string> GetScopesForRole(string role) => role switch
    {
        "SYSTEM_ADMIN" => new[] { "key:manage", "app:manage", "user:manage", "system:manage", "audit:read" },
        "KEY_ADMIN" => new[] { "key:manage", "key:read", "audit:read" },
        "APP_ADMIN" => new[] { "app:manage", "app:read" },
        "SECURITY_AUDITOR" => new[] { "audit:read", "security:read" },
        _ => Array.Empty<string>()
    };
}

public sealed record JwtSettings(string Issuer, int ExpiresInSeconds, string SigningKey);
```

## 38.3 应用凭据签名认证服务

```csharp
namespace CryptoPlatform.Authentication;

public interface IAppAuthenticationService
{
    Task<AppAuthResult> ValidateAsync(string appId, string? timestamp, string? nonce, string? signature, HttpRequest request);
}

public sealed record AppAuthResult(bool Success, string? ErrorCode = null, string? ErrorMessage = null);

public sealed class AppAuthenticationService : IAppAuthenticationService
{
    private readonly CryptoPlatformDbContext _db;
    private readonly IDistributedCache _redis;
    private readonly int _timestampToleranceSec = 300; // 5 分钟
    private readonly int _maxFailCount = 5;
    private readonly ILogger<AppAuthenticationService> _logger;

    public AppAuthenticationService(
        CryptoPlatformDbContext db, IDistributedCache redis,
        ILogger<AppAuthenticationService> logger)
    {
        _db = db; _redis = redis; _logger = logger;
    }

    public async Task<AppAuthResult> ValidateAsync(string appId, string? timestamp, string? nonce, string? signature, HttpRequest request)
    {
        // 1. 查询应用
        var app = await _db.Applications.FirstOrDefaultAsync(a => a.AppId == appId);
        if (app is null) return new(false, "AUTH_APP_NOT_FOUND", "应用不存在");
        if (app.Status != 0) return new(false, "AUTH_APP_DISABLED", "应用已禁用");

        // 2. IP 白名单
        if (!string.IsNullOrEmpty(app.IpWhitelist))
        {
            var clientIp = request.HttpContext.Connection.RemoteIpAddress?.ToString();
            if (!IsIpAllowed(clientIp, app.IpWhitelist))
                return new(false, "AUTH_IP_BLOCKED", "IP 不在白名单中");
        }

        // 3. 检查锁定状态
        var locked = await _redis.GetStringAsync($"app:locked:{appId}");
        if (locked is not null) return new(false, "AUTH_APP_LOCKED", "应用已锁定，请联系管理员解锁");

        // 4. Timestamp 校验
        if (string.IsNullOrEmpty(timestamp) || !long.TryParse(timestamp, out var ts))
            return new(false, "AUTH_TIMESTAMP_INVALID", "时间戳缺失或格式错误");
        var diff = Math.Abs(DateTimeOffset.UtcNow.ToUnixTimeSeconds() - ts);
        if (diff > _timestampToleranceSec)
            return new(false, "AUTH_TIMESTAMP_EXPIRED", "时间戳偏差过大");

        // 5. Nonce 防重放
        if (string.IsNullOrEmpty(nonce))
            return new(false, "AUTH_NONCE_MISSING", "Nonce 缺失");
        var nonceKey = $"nonce:{appId}:{nonce}";
        var exists = await _redis.GetStringAsync(nonceKey);
        if (exists is not null) return new(false, "AUTH_NONCE_REPLAYED", "Nonce 已使用");
        await _redis.SetStringAsync(nonceKey, "1",
            new DistributedCacheEntryOptions { AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(10) });

        // 6. 签名校验
        var activeSecret = await _db.ApplicationSecrets
            .Where(s => s.ApplicationId == app.Id && s.Status == 0)
            .OrderByDescending(s => s.SecretVersion)
            .FirstOrDefaultAsync();
        if (activeSecret is null) return new(false, "AUTH_NO_SECRET", "应用无有效凭据");

        var bodyBytes = await ReadBodyAsync(request);
        var bodyHash = Convert.ToBase64String(SM3Hash(bodyBytes));
        var stringToSign = $"{request.Method}\n{request.Path}\n{timestamp}\n{nonce}\n{bodyHash}";

        // 使用存储的 secret_hash 无法直接验签 → 需要存储明文 secret 的加密版本
        // 实际实现中，AppSecret 验证需使用可逆加密存储或维护验证密钥
        // 此处为设计骨架，实际需根据密钥管理策略调整
        var expectedSig = ComputeHmacSm3(stringToSign, activeSecret.SecretHash);
        if (!string.Equals(expectedSig, signature, StringComparison.Ordinal))
        {
            await RecordFailAsync(appId);
            return new(false, "AUTH_SIGNATURE_INVALID", "签名校验失败");
        }

        // 7. 更新最近使用时间
        activeSecret.LastUsedAt = DateTime.UtcNow;
        await _db.SaveChangesAsync();

        return new(true);
    }

    private async Task RecordFailAsync(string appId)
    {
        var key = $"app:fail:{appId}";
        var count = await _redis.GetStringAsync(key);
        var newCount = (int.TryParse(count, out var c) ? c : 0) + 1;
        await _redis.SetStringAsync(key, newCount.ToString(),
            new DistributedCacheEntryOptions { AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(10) });
        if (newCount >= _maxFailCount)
            await _redis.SetStringAsync($"app:locked:{appId}", "1",
                new DistributedCacheEntryOptions { AbsoluteExpirationRelativeToNow = TimeSpan.FromHours(1) });
    }

    private static bool IsIpAllowed(string? clientIp, string whitelist)
    {
        if (string.IsNullOrEmpty(clientIp)) return false;
        var allowed = whitelist.Split(',', StringSplitOptions.TrimEntries);
        return allowed.Any(a => a == clientIp || MatchCidr(clientIp, a));
    }

    private static bool MatchCidr(string ip, string cidr) => false; // TODO: CIDR 匹配实现

    private static async Task<byte[]> ReadBodyAsync(HttpRequest request)
    {
        request.EnableBuffering();
        using var ms = new MemoryStream();
        await request.Body.CopyToAsync(ms);
        request.Body.Position = 0;
        return ms.ToArray();
    }

    private static byte[] SM3Hash(byte[] data) => Array.Empty<byte>(); // TODO: SM3 实现
    private static string ComputeHmacSm3(string data, string key) => ""; // TODO: HMAC-SM3 实现
}
```

## 38.4 密钥授权检查器

```csharp
namespace CryptoPlatform.Authorization;

public interface IKeyAuthorizationChecker
{
    Task CheckAsync(string keyId, string callerAppId, string requiredPermission, CancellationToken ct);
}

public sealed class KeyAuthorizationChecker : IKeyAuthorizationChecker
{
    private readonly CryptoPlatformDbContext _db;

    public KeyAuthorizationChecker(CryptoPlatformDbContext db) => _db = db;

    public async Task CheckAsync(string keyId, string callerAppId, string requiredPermission, CancellationToken ct)
    {
        var key = await _db.Keys.FirstOrDefaultAsync(k => k.KeyId == keyId, ct);
        if (key is null) throw new BusinessException("KEY_NOT_FOUND", "密钥不存在");

        // Owner 直接放行
        if (key.OwnerAppId == callerAppId) return;

        // 检查显式授权
        var auth = await _db.KeyAuthorizations
            .FirstOrDefaultAsync(a => a.KeyId == keyId && a.AppId == callerAppId && a.Status == "ACTIVE", ct);

        if (auth is null)
            throw new BusinessException("KEY_PERMISSION_DENIED", "无权访问此密钥");

        if (auth.ExpiresAt.HasValue && auth.ExpiresAt < DateTime.UtcNow)
            throw new BusinessException("KEY_AUTH_EXPIRED", "授权已过期");

        var permissions = auth.Permissions.Split(',', StringSplitOptions.TrimEntries);
        if (!permissions.Contains(requiredPermission))
            throw new BusinessException("KEY_PERMISSION_DENIED", $"无 {requiredPermission} 权限");
    }
}
```

## 38.5 DI 注册

```csharp
public static class AuthServiceCollectionExtensions
{
    public static IServiceCollection AddAuthenticationServices(this IServiceCollection services, IConfiguration config)
    {
        services.Configure<JwtSettings>(config.GetSection("Jwt"));
        services.AddScoped<IAdminTokenService, AdminTokenService>();
        services.AddScoped<IAppAuthenticationService, AppAuthenticationService>();
        services.AddScoped<IKeyAuthorizationChecker, KeyAuthorizationChecker>();
        return services;
    }
}
```
